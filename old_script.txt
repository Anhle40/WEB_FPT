    function addPost() {
        const txt = document.getElementById('post-content').value.trim();
        const scope = document.getElementById('post-scope').value || 'local';
        if(!txt) return;
        // read image if selected
        const fileInput = document.getElementById('post-image');
        let imgUrl = "";
        const doPush = () => pushPost(txt, imgUrl, scope);
        if(fileInput && fileInput.files && fileInput.files[0]) {
            const file = fileInput.files[0];
            const reader = new FileReader();
            reader.onload = function(e) {
                imgUrl = e.target.result;
                doPush();
            };
            reader.readAsDataURL(file);
        } else {
            doPush();
        }
    }

        document.getElementById('post-modal').classList.add('hidden');
        // refresh feed based on scope selected
        if(scope === 'global') filterFeed('global'); else filterFeed('local');
    }

    function addEisenhowerTask() {
        const nameInput = document.getElementById('task-input-name');
        const deadlineInput = document.getElementById('task-input-deadline');
        const taskName = nameInput.value.trim();
        const deadline = deadlineInput.value ? new Date(deadlineInput.value).getTime() : null;
        
        if (!taskName) {
            alert('Vui l├▓ng nhß║¡p t├¬n c├┤ng viß╗çc!');
            return;
        }
        
        if (currentImportance === null || currentUrgency === null) {
            alert('Vui l├▓ng chß╗ìn mß╗⌐c ─æß╗Ö quan trß╗ìng v├á gß║Ñp g├íp!');
            return;
        }
        
        const task = {
            id: Date.now(),
            name: taskName,
            important: currentImportance,
            urgent: currentUrgency,
            createdAt: new Date().toLocaleDateString('vi-VN'),
            due: deadline,
            notified3h: false,
            notified1h: false
        };
        
        eisenhowerTasks.push(task);
        saveEisenhowerTasks();
        
        // Clear input
        nameInput.value = '';
        deadlineInput.value = '';
        currentImportance = null;
        currentUrgency = null;
        
        // Reset buttons
        document.getElementById('imp-btn-0').classList.add('border-slate-300', 'text-slate-700');
        document.getElementById('imp-btn-0').classList.remove('bg-slate-200', 'border-slate-500', 'text-slate-800');
        document.getElementById('imp-btn-1').classList.remove('bg-blue-100', 'border-blue-500', 'text-blue-700');
        document.getElementById('imp-btn-1').classList.add('border-slate-300', 'text-slate-700');
        document.getElementById('urg-btn-0').classList.add('border-slate-300', 'text-slate-700');
        document.getElementById('urg-btn-0').classList.remove('bg-slate-200', 'border-slate-500', 'text-slate-800');
        document.getElementById('urg-btn-1').classList.remove('bg-red-100', 'border-red-500', 'text-red-700');
        document.getElementById('urg-btn-1').classList.add('border-slate-300', 'text-slate-700');
        
        renderEisenhowerMatrix();
        checkDeadlines();
        saveHistory();
    }
  
    function deleteEisenhowerTask(id) {
        eisenhowerTasks = eisenhowerTasks.filter(t => t.id !== id);
        saveEisenhowerTasks();
        renderEisenhowerMatrix();
    }
  
    function renderEisenhowerMatrix() {
        // Clear all quadrants
        ['quad-1', 'quad-2', 'quad-3', 'quad-4'].forEach(qid => {
            document.getElementById(qid).innerHTML = '';
        });
        
        eisenhowerTasks.forEach(task => {
            let quadId;
            if (task.urgent && task.important) {
                quadId = 'quad-1';
            } else if (!task.urgent && task.important) {
                quadId = 'quad-2';
            } else if (task.urgent && !task.important) {
                quadId = 'quad-3';
            } else {
                quadId = 'quad-4';
            }
            
            const taskEl = document.createElement('div');
            taskEl.className = 'bg-white border-l-4 border-slate-400 rounded-lg p-3 flex justify-between items-start gap-2 shadow-sm hover:shadow-md transition';
            
            // Determine border color
            if (quadId === 'quad-1') {
                taskEl.classList.add('border-l-red-500', 'bg-red-50');
            } else if (quadId === 'quad-2') {
                taskEl.classList.add('border-l-blue-500', 'bg-blue-50');
            } else if (quadId === 'quad-3') {
                taskEl.classList.add('border-l-yellow-500', 'bg-yellow-50');
            } else {
                taskEl.classList.add('border-l-slate-400', 'bg-slate-50');
            }
            
            taskEl.innerHTML = `
                <div class="flex-1">
                    <p class="font-semibold text-slate-800 text-sm">${task.name}</p>
                    <p class="text-xs text-slate-500 mt-1">${task.createdAt}</p>
                </div>
                <button onclick="deleteEisenhowerTask(${task.id})" class="text-red-500 hover:bg-red-100 p-1.5 rounded transition flex-shrink-0">
                    <i class="fa-solid fa-trash text-sm"></i>
                </button>
            `;
            
            document.getElementById(quadId).appendChild(taskEl);
        });
        
        // Show empty state if no tasks
        ['quad-1', 'quad-2', 'quad-3', 'quad-4'].forEach(qid => {
            if (document.getElementById(qid).children.length === 0) {
                const emptyEl = document.createElement('div');
                emptyEl.className = 'text-center text-slate-400 text-sm py-8';
                emptyEl.innerText = 'Ch╞░a c├│ c├┤ng viß╗çc';
                document.getElementById(qid).appendChild(emptyEl);
            }
        });
    }
