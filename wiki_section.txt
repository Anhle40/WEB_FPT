  function saveHistory(){

  function undo(){
      if(historyIndex > 0){
          historyIndex--;
          applyHistory(historyStack[historyIndex]);
      }
  }
  function redo(){
      if(historyIndex < historyStack.length - 1){
          historyIndex++;
          applyHistory(historyStack[historyIndex]);
      }
  }
